# 测斜管理系统完整数据库表结构

## 一、用户与权限管理模块

### 1. 用户表 (users)

```
CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,      -- 用户名
    password_hash VARCHAR(255) NOT NULL,       -- 密码哈希
    real_name VARCHAR(50),                     -- 真实姓名
    phone VARCHAR(20),                         -- 联系电话
    email VARCHAR(100),                        -- 邮箱
    status ENUM('active', 'inactive', 'locked') DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

```

### 2. 角色表 (roles)

```
CREATE TABLE roles (
    id INT PRIMARY KEY AUTO_INCREMENT,
    role_code VARCHAR(50) UNIQUE NOT NULL,     -- 角色代码
    role_name VARCHAR(100) NOT NULL,           -- 角色名称
    role_type ENUM('system', 'project') DEFAULT 'system', -- 角色类型
    description VARCHAR(500),                  -- 角色描述
    is_default BOOLEAN DEFAULT FALSE,          -- 是否为默认角色
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

```

### 3. 用户角色关联表 (user\_roles)

```
CREATE TABLE user_roles (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    role_id INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY uk_user_role (user_id, role_id),
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (role_id) REFERENCES roles(id) ON DELETE CASCADE
);

```

### 4. 权限表 (permissions)

```
CREATE TABLE permissions (
    id INT PRIMARY KEY AUTO_INCREMENT,
    permission_code VARCHAR(100) UNIQUE NOT NULL, -- 权限代码
    permission_name VARCHAR(200) NOT NULL,     -- 权限名称
    permission_type ENUM('system', 'project', 'data') DEFAULT 'system',
    description VARCHAR(500),                  -- 权限描述
    module VARCHAR(50),                        -- 所属模块
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

```

### 5. 角色权限关联表 (role\_permissions)

```
CREATE TABLE role_permissions (
    id INT PRIMARY KEY AUTO_INCREMENT,
    role_id INT NOT NULL,
    permission_id INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY uk_role_permission (role_id, permission_id),
    FOREIGN KEY (role_id) REFERENCES roles(id) ON DELETE CASCADE,
    FOREIGN KEY (permission_id) REFERENCES permissions(id) ON DELETE CASCADE
);

```

### 6. 项目成员表 (project\_members)

```
CREATE TABLE project_members (
    id INT PRIMARY KEY AUTO_INCREMENT,
    project_id INT NOT NULL,                   -- 项目ID
    user_id INT NOT NULL,                      -- 用户ID
    project_role ENUM('manager', 'member', 'viewer') NOT NULL, -- 项目角色
    join_date DATE NOT NULL,                   -- 加入日期
    leave_date DATE,                           -- 离开日期
    is_active BOOLEAN DEFAULT TRUE,            -- 是否活跃
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    UNIQUE KEY uk_project_user (project_id, user_id),
    FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

```

## 二、设备管理模块

### 7. 设备表 (devices)

```
CREATE TABLE devices (
    id INT PRIMARY KEY AUTO_INCREMENT,
    device_id VARCHAR(50) UNIQUE NOT NULL,     -- 设备唯一ID（DTU ID）
    device_name VARCHAR(100),                  -- 设备名称
    
    -- 设备基本信息
    device_type VARCHAR(50),                   -- 设备类型：测斜仪、采集仪等
    manufacturer VARCHAR(100),                 -- 生产厂家
    model VARCHAR(50),                         -- 设备型号
    serial_number VARCHAR(100),                -- 序列号
    purchase_date DATE,                        -- 购买日期
    
    -- 通信与网络信息
    dtu_id VARCHAR(50),                        -- DTU ID（与device_id可能相同）
    network_status ENUM('online', 'offline') DEFAULT 'offline',
    last_online_time TIMESTAMP,                -- 最后在线时间
    ip_address VARCHAR(45),                    -- IP地址
    communication_protocol VARCHAR(50),        -- 通信协议
    signal_strength DECIMAL(6,2),              -- 信号强度
    
    -- 探头参数
    K_value DECIMAL(10,6),                     -- 标定系数K
    beta_value DECIMAL(10,6),                  -- 标定系数β
    calibration_date DATE,                     -- 标定日期
    calibration_org VARCHAR(100),              -- 标定机构
    valid_until DATE,                          -- 有效期至
    calibration_status ENUM('valid', 'expired', 'pending') DEFAULT 'valid',
    
    -- 设备状态
    status ENUM('idle', 'measuring', 'maintenance', 'fault', 'calibrating', 'charging') DEFAULT 'idle',
    battery_level DECIMAL(5,2),                -- 电池电量
    temperature DECIMAL(6,2),                  -- 设备温度
    
    -- 当前关联信息
    current_hole_id INT,                       -- 当前绑定的测孔ID
    current_project_id INT,                    -- 当前所在项目ID
    
    -- 统计信息
    total_usage_hours DECIMAL(10,2) DEFAULT 0, -- 总使用时长（小时）
    total_measurement_count INT DEFAULT 0,     -- 总测量次数
    total_measurement_depth DECIMAL(12,2) DEFAULT 0, -- 总测量深度（米）
    
    -- 系统信息
    created_by INT,                            -- 录入人
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (current_hole_id) REFERENCES monitoring_holes(id),
    FOREIGN KEY (current_project_id) REFERENCES projects(id),
    FOREIGN KEY (created_by) REFERENCES users(id)
);

```

### 8. 设备使用记录表 (device\_usage\_logs)

```
CREATE TABLE device_usage_logs (
    id INT PRIMARY KEY AUTO_INCREMENT,
    device_id INT NOT NULL,                    -- 设备ID
    
    -- 使用信息
    user_id INT NOT NULL,                      -- 使用人
    project_id INT NOT NULL,                   -- 使用项目
    hole_id INT NOT NULL,                      -- 测孔ID
    start_time TIMESTAMP NOT NULL,             -- 开始使用时间
    end_time TIMESTAMP,                        -- 结束使用时间
    
    -- 测量统计
    measurement_count INT DEFAULT 0,           -- 测量孔次
    total_measurement_depth DECIMAL(12,2),     -- 总测量深度（米）
    total_measurement_time INT,                -- 总测量时长（分钟）
    average_depth_per_measurement DECIMAL(8,2), -- 平均测量深度
    
    -- 设备状态记录
    start_battery_level DECIMAL(5,2),          -- 开始电池电量
    end_battery_level DECIMAL(5,2),            -- 结束电池电量
    start_temperature DECIMAL(6,2),            -- 开始温度
    end_temperature DECIMAL(6,2),              -- 结束温度
    start_signal_strength DECIMAL(6,2),        -- 开始信号强度
    end_signal_strength DECIMAL(6,2),          -- 结束信号强度
    
    -- 系统信息
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (device_id) REFERENCES devices(id),
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (project_id) REFERENCES projects(id),
    FOREIGN KEY (hole_id) REFERENCES monitoring_holes(id),
    
    INDEX idx_usage_time (start_time, end_time),
    INDEX idx_user_device (user_id, device_id)
);

```

### 9. 设备标定记录表 (device\_calibration\_logs)

```
CREATE TABLE device_calibration_logs (
    id INT PRIMARY KEY AUTO_INCREMENT,
    device_id INT NOT NULL,                    -- 设备ID
    calibration_date DATE NOT NULL,            -- 标定日期
    calibration_org VARCHAR(100) NOT NULL,     -- 标定机构
    calibrator VARCHAR(50),                    -- 标定人员
    
    -- 标定参数
    K_value DECIMAL(10,6) NOT NULL,           -- 标定系数K
    beta_value DECIMAL(10,6) NOT NULL,        -- 标定系数β
    temperature DECIMAL(6,2),                  -- 标定温度
    humidity DECIMAL(6,2),                     -- 标定湿度
    
    -- 标定数据
    calibration_data JSON,                     -- 标定原始数据
    calibration_points INT,                    -- 标定点数
    standard_deviation DECIMAL(10,6),          -- 标准差
    correlation_coefficient DECIMAL(6,4),      -- 相关系数
    
    -- 证书信息
    certificate_number VARCHAR(100),           -- 证书编号
    certificate_file VARCHAR(500),             -- 证书文件路径
    valid_until DATE,                          -- 有效期至
    
    -- 状态
    calibration_status ENUM('passed', 'failed', 'pending') DEFAULT 'passed',
    review_status ENUM('pending', 'approved', 'rejected') DEFAULT 'pending',
    
    -- 系统信息
    created_by INT,                            -- 记录人
    reviewed_by INT,                           -- 审核人
    reviewed_at TIMESTAMP,                     -- 审核时间
    review_notes TEXT,                         -- 审核意见
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (device_id) REFERENCES devices(id),
    FOREIGN KEY (created_by) REFERENCES users(id),
    FOREIGN KEY (reviewed_by) REFERENCES users(id)
);

```

## 四、测孔管理模块

### 12. 监测孔表 (monitoring\_holes)
```
CREATE TABLE monitoring_holes (
    id INT PRIMARY KEY AUTO_INCREMENT,
    hole_number VARCHAR(50) NOT NULL,          -- 测斜孔编号
    project_id INT NOT NULL,                   -- 所属项目
    hole_name VARCHAR(100),                    -- 测孔名称
    hole_type ENUM('retaining_structure', 'soil_body') NOT NULL, -- 测孔类型
    location_description VARCHAR(500),         -- 位置描述
    coordinates JSON,                          -- 坐标信息（经纬度）
    
    -- 测量参数
    depth DECIMAL(8,2) NOT NULL,               -- 孔深（米）
    measurement_interval DECIMAL(6,2) DEFAULT 0.5, -- 测量间隔（米）
    initial_measurement_distance DECIMAL(6,3) DEFAULT 0.05, -- 初始测量距离（米），对应设备参数 fMonitorFromStartDis
    calculation_reference ENUM('top', 'bottom') DEFAULT 'bottom', -- 起算点
    warning_standard DECIMAL(10,2),            -- 预警标准（mm）
    collection_method ENUM('manual', 'auto') DEFAULT 'manual', -- 采集方式
    
    -- 设备绑定信息
    device_id INT,                             -- 当前绑定的设备ID
    
    -- 定时器设置（自动采集时使用）
    timer_enabled BOOLEAN DEFAULT FALSE,       -- 定时器是否启用
    timer_type ENUM('interval', 'fixed_time') DEFAULT 'interval', -- 定时器类型
    timer_interval_minutes INT,                -- 采集间隔（分钟）
    fixed_times JSON,                          -- 固定采集时间点（JSON数组）
    timer_start_date DATE,                     -- 定时开始日期
    timer_end_date DATE,                       -- 定时结束日期
    
    -- 状态信息
    status ENUM('active', 'inactive', 'completed') DEFAULT 'active',
    last_measurement_time TIMESTAMP,           -- 最后测量时间
    next_scheduled_time TIMESTAMP,             -- 下次计划测量时间
    
    -- 统计信息
    total_measurements INT DEFAULT 0,          -- 总测量次数
    warning_count INT DEFAULT 0,               -- 预警次数
    
    -- 系统信息
    created_by INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE,
    FOREIGN KEY (device_id) REFERENCES devices(id),
    FOREIGN KEY (created_by) REFERENCES users(id),
    
    UNIQUE KEY uk_hole_project (hole_number, project_id)
);

```

### 13. 测孔绑定历史表 (hole\_binding\_history)

```
CREATE TABLE hole_binding_history (

    -- 绑定信息

    unbind_time TIMESTAMP,                     -- 解绑时间
    bind_type ENUM('manual', 'auto', 'scheduled') DEFAULT 'manual', -- 绑定类型
    bind_reason VARCHAR(200),                  -- 绑定原因
    unbind_reason VARCHAR(200),                -- 解配原因
    
    -- 操作信息
    bind_operator INT,                         -- 绑定操作人
    unbind_operator INT,                       -- 解绑操作人
    
    -- 绑定期间统计
    measurement_count INT DEFAULT 0,           -- 测量次数
    total_measurement_time INT DEFAULT 0,      -- 总测量时长（分钟）
    
    -- 系统信息
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (hole_id) REFERENCES monitoring_holes(id),
    FOREIGN KEY (device_id) REFERENCES devices(id),
    FOREIGN KEY (bind_operator) REFERENCES users(id),
    FOREIGN KEY (unbind_operator) REFERENCES users(id),
    
    INDEX idx_binding_time (bind_time, unbind_time),
    INDEX idx_hole_binding (hole_id, bind_time)
);

```

## 六、原始数据管理模块

### 16. 原始测量数据表 (raw\_measurement\_data)

```
CREATE TABLE raw_measurement_data (
    id INT PRIMARY KEY AUTO_INCREMENT,
    measurement_uuid VARCHAR(36) UNIQUE NOT NULL,
    
    -- 关联信息
    hole_id INT NOT NULL,                      -- 测孔ID
    device_id INT NOT NULL,                    -- 设备ID
    
    -- 测量信息
    measurement_time TIMESTAMP NOT NULL,       -- 测量时间
    measurement_mode ENUM('manual', 'auto', 'scheduled') NOT NULL, -- 测量模式
    operator_id INT,                           -- 操作人（手动测量）
    batch_number VARCHAR(50),                  -- 测量批次号
    
    -- 测量参数
    depth DECIMAL(8,2) NOT NULL,               -- 测量深度
    positive_value DECIMAL(12,6) NOT NULL,     -- 正测AD值
    negative_value DECIMAL(12,6) NOT NULL,     -- 反测AD值
    
    -- 环境参数
    temperature DECIMAL(6,2),                  -- 温度
    battery_voltage DECIMAL(6,2),              -- 电池电压
    signal_strength DECIMAL(6,2),              -- 信号强度
    
    -- 状态信息
    data_status ENUM('raw', 'processed', 'invalid', 'warning') DEFAULT 'raw',
    quality_flag ENUM('good', 'fair', 'poor') DEFAULT 'good',
    upload_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP, -- 上传时间
    processed_time TIMESTAMP,                  -- 处理时间
    
    -- 系统信息
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (hole_id) REFERENCES monitoring_holes(id),
    FOREIGN KEY (device_id) REFERENCES devices(id),
    FOREIGN KEY (operator_id) REFERENCES users(id),
    
    INDEX idx_measurement_time (measurement_time),
    INDEX idx_hole_depth (hole_id, depth),
    INDEX idx_device_measurement (device_id, measurement_time)
);

```

## 七、成果数据管理模块

### 18. 处理结果数据表 (processed\_result\_data)

```
CREATE TABLE processed_result_data (
    id INT PRIMARY KEY AUTO_INCREMENT,
    raw_data_id INT NOT NULL,                  -- 原始数据ID
    hole_id INT NOT NULL,                      -- 测孔ID
    device_id INT NOT NULL,                    -- 设备ID
    measurement_time TIMESTAMP NOT NULL,       -- 测量时间
    depth DECIMAL(8,2) NOT NULL,               -- 深度
    
    -- 计算中间值
    Y_average DECIMAL(12,6) NOT NULL,          -- Y平均值 = (正测值-反测值)/2
    inclination_angle DECIMAL(10,6) NOT NULL,  -- 倾角α = arcsin(Y平/K) - β
    depth_offset DECIMAL(10,4) NOT NULL,       -- 深度偏移值Δ = 500 * sin(α)
    
    -- 变形计算结果
    current_pipe_value DECIMAL(12,4),          -- 本次管形值
    initial_pipe_value DECIMAL(12,4),          -- 初始管形值
    cumulative_displacement DECIMAL(12,4),     -- 累计变形值
    previous_pipe_value DECIMAL(12,4),         -- 上次管形值
    current_displacement DECIMAL(12,4),        -- 当次变形值
    deformation_rate DECIMAL(12,4),            -- 变形速率
    
    -- 预警信息
    warning_level ENUM('normal', 'warning', 'alarm') DEFAULT 'normal',
    warning_message VARCHAR(200),
    
    processed_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (raw_data_id) REFERENCES raw_measurement_data(id),
    FOREIGN KEY (hole_id) REFERENCES monitoring_holes(id),
    FOREIGN KEY (device_id) REFERENCES devices(id),
    UNIQUE KEY uk_raw_data (raw_data_id)
);

```

## 八、系统配置与日志模块

### 21. 系统配置表 (system\_config)


```

    u.id AS user_id,
    u.username,
    u.real_name,
    pm.project_id,
    p.project_name,
    pm.project_role,
    GROUP_CONCAT(DISTINCT perm.permission_name ORDER BY perm.permission_name) AS permissions_list,
    CASE 
        WHEN pm.project_role = 'manager' THEN 'full_access'
        WHEN pm.project_role = 'member' THEN 'operational_access'
        WHEN pm.project_role = 'viewer' THEN 'read_only'
    END AS access_level
FROM users u
LEFT JOIN project_members pm ON u.id = pm.user_id AND pm.is_active = TRUE
LEFT JOIN projects p ON pm.project_id = p.id
LEFT JOIN project_permission_view ppv ON pm.project_id = ppv.project_id AND pm.user_id = ppv.user_id
LEFT JOIN permissions perm ON ppv.permission_id = perm.id
GROUP BY u.id, pm.project_id, pm.project_role;
```

## 十一、权限验证函数

```
DELIMITER //
CREATE FUNCTION check_user_permission(
    p_user_id INT,
    p_permission_code VARCHAR(100),
    p_project_id INT DEFAULT NULL
) RETURNS BOOLEAN
READS SQL DATA
BEGIN
    DECLARE has_permission BOOLEAN DEFAULT FALSE;
    
    -- 如果是系统管理员，拥有所有权限
    IF EXISTS (
        SELECT 1 FROM user_roles ur
        JOIN roles r ON ur.role_id = r.id
        WHERE ur.user_id = p_user_id AND r.role_code = 'system_admin'
    ) THEN
        RETURN TRUE;
    END IF;
    
    -- 检查项目权限
    IF p_project_id IS NOT NULL THEN
        -- 检查用户在项目中的角色权限
        SELECT COUNT(*) > 0 INTO has_permission
        FROM project_permission_view ppv
        WHERE ppv.user_id = p_user_id
          AND ppv.project_id = p_project_id
          AND ppv.permission_code = p_permission_code;
    ELSE
        -- 检查系统权限
        SELECT COUNT(*) > 0 INTO has_permission
        FROM user_roles ur
        JOIN role_permissions rp ON ur.role_id = rp.role_id
        JOIN permissions p ON rp.permission_id = p.id
        WHERE ur.user_id = p_user_id
          AND p.permission_code = p_permission_code;
    END IF;
    
    RETURN has_permission;
END //
DELIMITER ;

```
